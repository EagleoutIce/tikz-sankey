% \CheckInvariants{prefix}: (an empty prefix skips the node check, buckets have no node of their own) after \end{sankey}; every drawn node and every flow sits where it should
\ExplSyntaxOn
\dim_new:N \l_sankeytest_ext_dim
\cs_new_protected:Npn \sankeytest_check_nodes:n #1
  {
    \dim_zero:N \l_sankeytest_ext_dim
    \seq_map_inline:Nn \g__sankey_nodes_seq
      { \dim_set:Nn \l_sankeytest_ext_dim { \dim_max:nn { \l_sankeytest_ext_dim } { \sankeyNodeInfo {##1} { bottom } } } }
    \seq_map_inline:Nn \g__sankey_nodes_seq
      {
        \__sankey_has:nnF { dummy } {##1}
          {
            \fp_compare:nNnT { \sankeyNodeInfo {##1} { value } } > { 0 }
              {
                \tl_if_empty:nF {#1} { \TestHas { node~##1 } { #1 ##1 } }
                \dim_compare:nNnF { \sankeyNodeInfo {##1} { height } } > { 0pt }
                  { \msg_error:nneee { sankey-test } { fail } { height~##1 } { \sankeyNodeInfo {##1} { height } } { > ~ 0pt } }
                \dim_compare:nNnF { \sankeyNodeInfo {##1} { top } } > { -0.01pt }
                  { \msg_error:nneee { sankey-test } { fail } { top~##1 } { \sankeyNodeInfo {##1} { top } } { >= ~ 0pt } }
                \dim_compare:nNnF { \sankeyNodeInfo {##1} { bottom } } < { \l_sankeytest_ext_dim + 0.01pt }
                  { \msg_error:nneee { sankey-test } { fail } { bottom~##1 } { \sankeyNodeInfo {##1} { bottom } } { <= ~ \dim_eval:n { \l_sankeytest_ext_dim } } }
              }
          }
      }
  }
\cs_new_protected:Npn \sankeytest_in:nnn #1#2#3 % label, dim, node
  {
    \dim_compare:nNnF {#2} > { \sankeyNodeInfo {#3} { top } - 0.05pt }
      { \msg_error:nneee { sankey-test } { fail } { #1~above } { \dim_eval:n {#2} } { >= ~ \sankeyNodeInfo {#3} { top } } }
    \dim_compare:nNnF {#2} < { \sankeyNodeInfo {#3} { bottom } + 0.05pt }
      { \msg_error:nneee { sankey-test } { fail } { #1~below } { \dim_eval:n {#2} } { <= ~ \sankeyNodeInfo {#3} { bottom } } }
  }
\cs_new_protected:Npn \sankeytest_check_flows:
  {
    \int_step_inline:nn { \sankeyFlowCount }
      {
        \__sankey_has:nnF { fdrop } {##1}
          {
            \fp_compare:nNnT { \sankeyFlowInfo {##1} { value } } > { 0 }
              {
                \sankeytest_in:nnn { flow~##1~start } { \sankeyFlowInfo {##1} { start } } { \sankeyFlowInfo {##1} { from } }
                \sankeytest_in:nnn { flow~##1~end } { \sankeyFlowInfo {##1} { end } } { \sankeyFlowInfo {##1} { to } }
              }
          }
      }
  }

\cs_new_protected:Npn \sankeytest_compare:nnnnnnnn #1#2#3#4#5#6#7#8
  {
    \bool_if:nT
      {
        % a label may cover its own node; nodes are checked by \CheckInvariants
        ! ( \str_if_eq_p:nn {#3} { node } && \str_if_eq_p:nn {#7} { node } )
        && ! ( \str_if_eq_p:nn {#2} {#6} && ! ( \str_if_eq_p:nn {#3} { label } && \str_if_eq_p:nn {#7} { label } ) )
      }
      {
        \sankeytest_hit:nnTF {#1} {#5}
          { \msg_error:nneee { sankey-test } { fail } { overlap } { #4 } { #8 } }
          { }
      }
  }

  }
